   <script src="https://cdnjs.cloudflare.com/ajax/libs/three.js/r128/three.min.js"></script>

   <script>
       /**
        * Creates a fully procedural Hawaiian Pōhinahina (Vitex rotundifolia) 3D Model.
        * Requirements: Self-contained, procedural realism, MeshStandardMaterial, no flatShading,
        * low-growing spreading shrub, opposite silvery-green leaves, creeping stems, clustered flowers.
        * @returns {THREE.Group} The complete Pōhinahina plant.
        */
       function createPohinahina() {
           const pohinahinaGroup = new THREE.Group();
           const dummy = new THREE.Object3D();
           const colorHelper = new THREE.Color();


           // --- CONFIGURATION ---
           const numMainStems = 18;          // Spreading main branches
           const subStemsPerMain = 3;        // Branching density to create a dense mound
           const leafPairsPerUnit = 6;       // Density of opposite leaf pairs along stems
          
           // --- MATERIALS ---
           // Woody/creeping stems, pale grey-brown
           const stemMat = new THREE.MeshStandardMaterial({
               color: 0x8a8175,
               roughness: 0.8,
               metalness: 0.0
           });


           // Pōhinahina leaves: Oval, silvery-green/gray, velvety texture
           // Achieved with high roughness and slight metallic sheen to mimic fine white hairs
           const leafMat = new THREE.MeshStandardMaterial({
               color: 0x8ba192,  // Silvery-sage green
               roughness: 0.9,   // Very diffuse/matte
               metalness: 0.1,   // Slight sheen catches grazing light like tiny hairs
               side: THREE.DoubleSide
           });


           // Small purple/lavender flowers typical of Vitex species
           const flowerMat = new THREE.MeshStandardMaterial({
               color: 0x826ca3,  // Lavender purple
               roughness: 0.6,
               metalness: 0.0
           });


           // --- 1. STEM GENERATION (Creeping, mounding habit) ---
           const stems = []; // Store curves for leaf placement
          
           for (let i = 0; i < numMainStems; i++) {
               // Radial spread with random chaos
               const angle = (i / numMainStems) * Math.PI * 2 + (Math.random() * 0.4);
               const length = 2.0 + Math.random() * 2.0; // Spreads 2 to 4 units out
              
               // Arching profile: starts at center, arches up slightly, droops to ground
               const p0 = new THREE.Vector3(0, 0, 0);
               const p1 = new THREE.Vector3(
                   Math.cos(angle) * length * 0.3,
                   0.4 + Math.random() * 0.6, // Arches up into a mound
                   Math.sin(angle) * length * 0.3
               );
               const p2 = new THREE.Vector3(
                   Math.cos(angle) * length * 0.7,
                   0.3 + Math.random() * 0.4,
                   Math.sin(angle) * length * 0.7
               );
               const p3 = new THREE.Vector3(
                   Math.cos(angle) * length,
                   0.05 + Math.random() * 0.15, // Creeps along the ground at the tips
                   Math.sin(angle) * length
               );
              
               const mainCurve = new THREE.CatmullRomCurve3([p0, p1, p2, p3]);
               stems.push({ curve: mainCurve, length: length, isMain: true });


               // Generate sub-branches stemming from the main branch
               for (let j = 0; j < subStemsPerMain; j++) {
                   const tStart = 0.2 + (j / subStemsPerMain) * 0.6 + (Math.random() * 0.1);
                   const startPt = mainCurve.getPointAt(tStart);
                  
                   const subLength = 1.0 + Math.random() * 1.5;
                   const spreadDir = Math.random() > 0.5 ? 1 : -1;
                   const subAngle = angle + spreadDir * (0.4 + Math.random() * 0.4);
                  
                   const sp1 = startPt.clone().add(new THREE.Vector3(
                       Math.cos(subAngle) * subLength * 0.5,
                       0.2 + Math.random() * 0.3,
                       Math.sin(subAngle) * subLength * 0.5
                   ));
                   const sp2 = startPt.clone().add(new THREE.Vector3(
                       Math.cos(subAngle) * subLength,
                       0.05 + Math.random() * 0.1,
                       Math.sin(subAngle) * subLength
                   ));
                  
                   const subCurve = new THREE.CatmullRomCurve3([startPt, sp1, sp2]);
                   stems.push({ curve: subCurve, length: subLength, isMain: false });
               }
           }


           // Create geometry for all stems
           stems.forEach(stemData => {
               const thickness = stemData.isMain ? 0.025 : 0.015;
               const stemGeo = new THREE.TubeGeometry(stemData.curve, 20, thickness, 6, false);
               const stemMesh = new THREE.Mesh(stemGeo, stemMat);
               stemMesh.castShadow = true;
               stemMesh.receiveShadow = true;
               pohinahinaGroup.add(stemMesh);
           });


           // --- 2. LEAF GENERATION (Oval, Silvery, Opposite Decussate Arrangement) ---
           const leafLength = 0.25;
           const leafGeo = new THREE.CylinderGeometry(0.01, 0.01, leafLength, 8, 4);
           leafGeo.translate(0, leafLength / 2, 0); // Origin at base
          
           // Deform the cylinder into a cupped, rounded oval shape (typical of Pōhinahina)
           const pos = leafGeo.attributes.position;
           for (let i = 0; i < pos.count; i++) {
               let y = pos.getY(i);
               let t = y / leafLength; // Normalized length (0 to 1)
              
               // Widen into a rounded oval/spoon shape
               let widthProfile = Math.sin(t * Math.PI);
               let x = pos.getX(i) * widthProfile * 12.0;
              
               // Flatten the Z axis to make it a leaf
               let z = pos.getZ(i) * 0.1;
              
               // Slightly cup the leaf upwards like a spoon
               z += Math.abs(x) * 0.15;
               z -= Math.sin(t * Math.PI) * 0.05;
              
               pos.setXYZ(i, x, y, z);
           }
           leafGeo.computeVertexNormals();


           // Calculate total leaves needed
           let totalLeaves = 0;
           stems.forEach(stem => {
               totalLeaves += Math.floor(stem.length * leafPairsPerUnit) * 2; // *2 because they grow in pairs
           });


           const leavesInstanced = new THREE.InstancedMesh(leafGeo, leafMat, totalLeaves);
           leavesInstanced.castShadow = true;
           leavesInstanced.receiveShadow = true;
           pohinahinaGroup.add(leavesInstanced);


           let globalLeafIdx = 0;


           // --- 3. FLOWER CLUSTER GENERATION ---
           // Small spherical purple clusters at the tips of branches
           const flowerGeo = new THREE.SphereGeometry(0.015, 6, 6);
           // Estimate max flowers (approx 15 flowers per stem tip on half the stems)
           const flowerInstanced = new THREE.InstancedMesh(flowerGeo, flowerMat, stems.length * 20);
           flowerInstanced.castShadow = true;
           pohinahinaGroup.add(flowerInstanced);
          
           let globalFlowerIdx = 0;


           // --- PLACEMENT LOGIC ---
           stems.forEach(stemData => {
               const numPairs = Math.floor(stemData.length * leafPairsPerUnit);
              
               for (let i = 1; i <= numPairs; i++) {
                   const t = i / (numPairs + 1);
                   const point = stemData.curve.getPointAt(t);
                   const tangent = stemData.curve.getTangentAt(t).normalize();
                   const up = new THREE.Vector3(0, 1, 0);
                  
                   // Local axes
                   const binormal = new THREE.Vector3().crossVectors(tangent, up).normalize();
                   const normal = new THREE.Vector3().crossVectors(binormal, tangent).normalize();


                   // Decussate arrangement: alternate pairs by 90 degrees
                   const angleOffset = (i % 2 === 0) ? 0 : Math.PI / 2;


                   // Leaf scale: smaller near the tips (new growth)
                   const sizeScale = THREE.MathUtils.lerp(1.2, 0.6, t);


                   // Slightly randomize color for older vs newer leaves
                   const hue = 0.38 + (Math.random() * 0.04 - 0.02); // Sage green base
                   const sat = 0.15 + (t * 0.05); // Grayish, slightly more color at tips
                   const light = 0.55 + (Math.random() * 0.1); // Bright, silvery
                   colorHelper.setHSL(hue, sat, light);


                   // Place two opposite leaves
                   const angles = [angleOffset, angleOffset + Math.PI];
                   angles.forEach(ang => {
                       if (globalLeafIdx >= totalLeaves) return;


                       // Rotate out from the stem
                       const leafOut = binormal.clone().applyAxisAngle(tangent, ang).normalize();
                      
                       // Angle forward along the stem and slightly upward
                       const leafDir = leafOut.clone()
                           .add(tangent.clone().multiplyScalar(0.6))
                           .add(new THREE.Vector3(0, 0.2, 0))
                           .normalize();


                       dummy.position.copy(point);
                       dummy.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), leafDir);
                      
                       // Add slight random twist to the leaf
                       dummy.rotateY((Math.random() - 0.5) * 0.2);
                      
                       dummy.scale.set(sizeScale, sizeScale, sizeScale);
                       dummy.updateMatrix();


                       leavesInstanced.setMatrixAt(globalLeafIdx, dummy.matrix);
                       leavesInstanced.setColorAt(globalLeafIdx, colorHelper.clone());
                       globalLeafIdx++;
                   });
               }


               // --- FLOWER CLUSTERS AT TIPS ---
               // Add flowers to the ends of about 70% of the stems
               if (Math.random() > 0.3) {
                   const tipPoint = stemData.curve.getPointAt(0.98);
                   const tipTangent = stemData.curve.getTangentAt(0.98).normalize();
                   const numFlowers = 8 + Math.floor(Math.random() * 10);
                  
                   for (let f = 0; f < numFlowers; f++) {
                       // Cluster radially around the tip, extending slightly forward
                       const fRadius = 0.03 + Math.random() * 0.04;
                       const fAngle = Math.random() * Math.PI * 2;
                       const fForward = (Math.random() - 0.2) * 0.08;
                      
                       // Calculate offset
                       const binormal = new THREE.Vector3().crossVectors(tipTangent, new THREE.Vector3(0,1,0)).normalize();
                       const normal = new THREE.Vector3().crossVectors(binormal, tipTangent).normalize();
                      
                       const offset = new THREE.Vector3()
                           .addScaledVector(binormal, Math.cos(fAngle) * fRadius)
                           .addScaledVector(normal, Math.sin(fAngle) * fRadius)
                           .addScaledVector(tipTangent, fForward);


                       dummy.position.copy(tipPoint).add(offset);
                       dummy.scale.setScalar(0.8 + Math.random() * 0.6);
                      
                       // Random rotation for the spherical bumps so lighting looks varied
                       dummy.rotation.set(Math.random()*Math.PI, Math.random()*Math.PI, Math.random()*Math.PI);
                       dummy.updateMatrix();


                       // Vary purple color slightly
                       colorHelper.setHSL(0.72 + Math.random()*0.05, 0.4 + Math.random()*0.2, 0.5 + Math.random()*0.1);
                      
                       flowerInstanced.setMatrixAt(globalFlowerIdx, dummy.matrix);
                       flowerInstanced.setColorAt(globalFlowerIdx, colorHelper.clone());
                       globalFlowerIdx++;
                   }
               }
           });


           leavesInstanced.instanceMatrix.needsUpdate = true;
           leavesInstanced.instanceColor.needsUpdate = true;
           flowerInstanced.instanceMatrix.needsUpdate = true;
           flowerInstanced.instanceColor.needsUpdate = true;


           return pohinahinaGroup;
       }


       // ==========================================
       // SCENE SETUP & PREVIEW ENVIRONMENT
       // ==========================================


       const container = document.getElementById('canvas-container');
       const scene = new THREE.Scene();
       scene.fog = new THREE.FogExp2(0xa0d8ef, 0.02); // Match coastal sky


       const camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 100);
       camera.position.set(6, 4, 6);


       const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
       renderer.setSize(window.innerWidth, window.innerHeight);
       renderer.setPixelRatio(window.devicePixelRatio);
       renderer.shadowMap.enabled = true;
       renderer.shadowMap.type = THREE.PCFSoftShadowMap;
       container.appendChild(renderer.domElement);


       const controls = new THREE.OrbitControls(camera, renderer.domElement);
       controls.enableDamping = true;
       controls.dampingFactor = 0.05;
       controls.target.set(0, 0.5, 0); // Focus on the center of the mound


       // --- LIGHTING ---
       // Bright coastal ambient bounce light (warm sand bouncing up)
       const ambientLight = new THREE.HemisphereLight(0xffffff, 0xd2b48c, 0.8);
       scene.add(ambientLight);


       // Strong, bright directional sunlight
       const dirLight = new THREE.DirectionalLight(0xfffdf0, 1.2);
       dirLight.position.set(5, 10, 4);
       dirLight.castShadow = true;
       dirLight.shadow.camera.top = 5;
       dirLight.shadow.camera.bottom = -5;
       dirLight.shadow.camera.left = -5;
       dirLight.shadow.camera.right = 5;
       dirLight.shadow.camera.near = 0.1;
       dirLight.shadow.camera.far = 25;
       dirLight.shadow.mapSize.width = 2048; // Crisp shadows for small leaves
       dirLight.shadow.mapSize.height = 2048;
       dirLight.shadow.bias = -0.0005;
       scene.add(dirLight);


       // --- ENVIRONMENT ---
       // Sandy coastal beach ground plane
       const groundGeo = new THREE.PlaneGeometry(40, 40);
       const groundMat = new THREE.MeshStandardMaterial({
           color: 0xd4c4a8, // Sand color
           roughness: 1.0,
           metalness: 0.0
       });
       const ground = new THREE.Mesh(groundGeo, groundMat);
       ground.rotation.x = -Math.PI / 2;
       ground.receiveShadow = true;
       scene.add(ground);


       // Add some small scattered beach rocks
       const rockGeo = new THREE.DodecahedronGeometry(0.2, 1);
       const rockMat = new THREE.MeshStandardMaterial({ color: 0x8a8581, roughness: 0.9 });
       for(let i = 0; i < 8; i++) {
           const rock = new THREE.Mesh(rockGeo, rockMat);
           const dist = 1.0 + Math.random() * 3.0;
           const angle = Math.random() * Math.PI * 2;
           rock.position.set(Math.cos(angle)*dist, -0.05, Math.sin(angle)*dist);
           rock.rotation.set(Math.random(), Math.random(), Math.random());
           rock.scale.set(1, Math.random()*0.5 + 0.3, 1);
           rock.castShadow = true;
           rock.receiveShadow = true;
           scene.add(rock);
       }


       // --- GENERATE AND ADD PŌHINAHINA ---
       console.log("Generating Procedural Pōhinahina...");
       const pohinahinaPlant = createPohinahina();
       scene.add(pohinahinaPlant);


       // --- ANIMATION LOOP ---
       function animate() {
           requestAnimationFrame(animate);
           controls.update();
           renderer.render(scene, camera);
       }


       // Handle Window Resize
       window.addEventListener('resize', () => {
           camera.aspect = window.innerWidth / window.innerHeight;
           camera.updateProjectionMatrix();
           renderer.setSize(window.innerWidth, window.innerHeight);
       });


       // Start render
       animate();


   </script>
