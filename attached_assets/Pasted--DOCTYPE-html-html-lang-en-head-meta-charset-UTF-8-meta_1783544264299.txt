       <h1>Procedural Palapalai Fern</h1>
       <p>100% Procedurally generated using mathematical curves.<br>
       Constructed from ~20,000 simple clustered geometries.<br>
       Left Click & Drag to rotate • Scroll to zoom</p>

   <script type="module">
       import * as THREE from 'three';
       import { OrbitControls } from 'three/addons/controls/OrbitControls.js';


       // --- SCENE SETUP ---
       const scene = new THREE.Scene();
       scene.fog = new THREE.FogExp2(0x1a1f18, 0.15);


       const camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 100);
       camera.position.set(0, 2, 4);


       const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
       renderer.setSize(window.innerWidth, window.innerHeight);
       renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
       renderer.shadowMap.enabled = true;
       renderer.shadowMap.type = THREE.PCFSoftShadowMap;
       document.body.appendChild(renderer.domElement);


       const controls = new OrbitControls(camera, renderer.domElement);
       controls.enableDamping = true;
       controls.dampingFactor = 0.05;
       controls.target.set(0, 0.8, 0);


       // --- LIGHTING ---
       const ambientLight = new THREE.AmbientLight(0x405040, 1.5); // Soft green-ish ambient
       scene.add(ambientLight);


       const sunLight = new THREE.DirectionalLight(0xfffae6, 3.5); // Warm sunlight
       sunLight.position.set(5, 8, 3);
       sunLight.castShadow = true;
       sunLight.shadow.mapSize.width = 2048;
       sunLight.shadow.mapSize.height = 2048;
       sunLight.shadow.camera.near = 0.5;
       sunLight.shadow.camera.far = 15;
       sunLight.shadow.camera.left = -3;
       sunLight.shadow.camera.right = 3;
       sunLight.shadow.camera.top = 3;
       sunLight.shadow.camera.bottom = -3;
       sunLight.shadow.bias = -0.001;
       scene.add(sunLight);


       const backLight = new THREE.DirectionalLight(0x7bb03b, 1.0); // Bounced leaf light
       backLight.position.set(-5, -2, -5);
       scene.add(backLight);


       // --- PROCEDURAL GENERATION ---
      
       /**
        * Generates the entire fern plant cluster procedurally.
        * Meets requirements: Self-contained, procedural, MeshStandardMaterial, clustered primitives.
        */
       function createProceduralFern() {
           const plantGroup = new THREE.Group();


           // 1. Define configuration for the clustering density
           const numFronds = 18; // Wide lacy cluster
           const branchesPerFrond = 28; // Density of the frond
           const leavesPerBranch = 14; // Density of the leaflets
           const fuzzPerFrond = 150; // Fuzzy hairy stems


           const totalBranches = numFronds * branchesPerFrond * 2; // * 2 for left/right
           const totalLeaves = totalBranches * leavesPerBranch * 2; // * 2 for left/right leaflets
           const totalFuzz = numFronds * fuzzPerFrond;


           // 2. Define Materials (No flat shading, natural colors)
           const stemMaterial = new THREE.MeshStandardMaterial({
               color: 0x4a3b2c, // Dark brownish-green base
               roughness: 0.95,
               metalness: 0.0
           });


           const leafMaterial = new THREE.MeshStandardMaterial({
               color: 0x6ca332, // Bright, healthy green
               roughness: 0.6,
               metalness: 0.05,
               side: THREE.DoubleSide
           });


           // 3. Construct Simple Geometries to cluster
          
           // Branch Geometry: A thin cylinder, translated so its origin is at the base
           const branchGeo = new THREE.CylinderGeometry(0.003, 0.001, 1, 4);
           branchGeo.translate(0, 0.5, 0); // Base to origin
           branchGeo.rotateX(Math.PI / 2); // Pointing forward along +Z


           // Leaf Geometry: A flattened cone representing a single pinnule (leaflet)
           const leafGeo = new THREE.ConeGeometry(0.012, 0.06, 4);
           leafGeo.translate(0, 0.03, 0); // Base to origin
           leafGeo.rotateY(Math.PI / 4); // Rotate 45deg to create a diamond cross-section
           leafGeo.scale(1, 1, 0.15); // Flatten heavily into a leaf shape
           leafGeo.rotateX(Math.PI / 2); // Pointing forward along +Z


           // Fuzz Geometry: Tiny, thin splinters
           const fuzzGeo = new THREE.CylinderGeometry(0.0005, 0.0005, 0.02, 3);
           fuzzGeo.translate(0, 0.01, 0);
           fuzzGeo.rotateX(Math.PI / 2);


           // 4. Setup Instanced Meshes for massive performance with clustering
           const branchMesh = new THREE.InstancedMesh(branchGeo, stemMaterial, totalBranches);
           const leafMesh = new THREE.InstancedMesh(leafGeo, leafMaterial, totalLeaves);
           const fuzzMesh = new THREE.InstancedMesh(fuzzGeo, stemMaterial, totalFuzz);
          
           branchMesh.castShadow = true; branchMesh.receiveShadow = true;
           leafMesh.castShadow = true; leafMesh.receiveShadow = true;


           let branchIdx = 0;
           let leafIdx = 0;
           let fuzzIdx = 0;


           const dummy = new THREE.Object3D();
           const upVector = new THREE.Vector3(0, 1, 0);


           // 5. Procedural Layout Loop
           for (let i = 0; i < numFronds; i++) {
               // Determine base stats for this individual frond
               const frondAngle = (i / numFronds) * Math.PI * 2 + (Math.random() * 0.2 - 0.1);
               const frondScale = 0.6 + Math.pow(Math.random(), 2) * 0.6; // Some small, some large
               const frondDroop = 0.5 + Math.random() * 0.7; // How far the tip bows down


               // Generate the graceful arching spine of the frond
               const curve = new THREE.QuadraticBezierCurve3(
                   new THREE.Vector3(0, 0, 0),
                   new THREE.Vector3(0, frondScale * 1.8, frondScale * 0.4), // Arch up
                   new THREE.Vector3(0, frondScale * (1.5 - frondDroop), frondScale * 1.8) // Arch out and down
               );


               // Rotate curve points to fan them out in a circle
               const rotMatrix = new THREE.Matrix4().makeRotationY(frondAngle);
               curve.v1.applyMatrix4(rotMatrix);
               curve.v2.applyMatrix4(rotMatrix);


               // Create the central solid stem for this frond
               const mainStemGeo = new THREE.TubeGeometry(curve, 32, 0.012, 5, false);
               const mainStem = new THREE.Mesh(mainStemGeo, stemMaterial);
               mainStem.castShadow = true;
               mainStem.receiveShadow = true;
               plantGroup.add(mainStem);


               // Add Fuzz (hair) to the base of the stem
               for (let f = 0; f < fuzzPerFrond; f++) {
                   const t = Math.pow(Math.random(), 2) * 0.35; // Heavily biased to the bottom 35% of stem
                   const pt = curve.getPointAt(t);
                   const tan = curve.getTangentAt(t);


                   dummy.position.copy(pt);
                  
                   // Generate a random vector perpendicular to the stem
                   const randomDir = new THREE.Vector3(Math.random() - 0.5, Math.random() - 0.5, Math.random() - 0.5).normalize();
                   const dot = randomDir.dot(tan);
                   randomDir.sub(tan.clone().multiplyScalar(dot)).normalize();


                   // Offset to surface of stem
                   dummy.position.add(randomDir.clone().multiplyScalar(0.012));
                  
                   // Point outward
                   dummy.lookAt(dummy.position.clone().add(randomDir));
                   dummy.scale.set(1, 1, Math.random() * 1.5 + 0.5);
                   dummy.updateMatrix();
                   fuzzMesh.setMatrixAt(fuzzIdx++, dummy.matrix);
               }


               // Add Branches (Pinnae) and Leaves (Pinnules)
               for (let b = 0; b < branchesPerFrond; b++) {
                   // Distribute along the stem, leaving the base bare
                   const t_branch = 0.15 + (b / branchesPerFrond) * 0.83;
                   const branchPos = curve.getPointAt(t_branch);
                   const branchTan = curve.getTangentAt(t_branch);


                   // Determine branch sideways orientation
                   let branchRight = new THREE.Vector3().crossVectors(branchTan, upVector).normalize();
                   if (branchRight.lengthSq() < 0.001) {
                       branchRight.set(Math.cos(frondAngle), 0, -Math.sin(frondAngle)).normalize();
                   }
                   const branchNormal = new THREE.Vector3().crossVectors(branchRight, branchTan).normalize();


                   // Branch length: long in the middle, tapering to a point at the end
                   const taper = Math.sin(Math.PI * (t_branch - 0.15) / 0.85);
                   const branchLength = ((1 - t_branch) * 0.7 + taper * 0.3) * frondScale * 0.45;


                   // Create left and right branches
                   for (let side = -1; side <= 1; side += 2) {
                       const dir = branchRight.clone().multiplyScalar(side);
                       dir.add(branchTan.clone().multiplyScalar(0.6)); // Sweep forward along stem
                       dir.add(branchNormal.clone().multiplyScalar(-0.15)); // Slight downward droop
                       dir.normalize();


                       const endPos = branchPos.clone().add(dir.clone().multiplyScalar(branchLength));


                       // 1. Matrix for Branch
                       dummy.position.copy(branchPos);
                       dummy.lookAt(endPos);
                       dummy.scale.set(1, 1, branchLength);
                       dummy.updateMatrix();
                       branchMesh.setMatrixAt(branchIdx++, dummy.matrix);


                       // 2. Matrices for Leaves on this branch
                       for (let l = 0; l < leavesPerBranch; l++) {
                           const t_leaf = (l + 0.5) / leavesPerBranch;
                           const leafBasePos = branchPos.clone().lerp(endPos, t_leaf);
                          
                           // Leaves taper in size towards the tip of the branch
                           const leafScale = (1 - t_leaf) * 0.7 + 0.3;


                           for (let lSide = -1; lSide <= 1; lSide += 2) {
                               const leafRight = new THREE.Vector3().crossVectors(dir, upVector).normalize();
                               const lDir = leafRight.clone().multiplyScalar(lSide);
                              
                               // Leaves point out, slightly forward, and open upwards to catch light
                               lDir.add(dir.clone().multiplyScalar(0.8));
                               lDir.add(upVector.clone().multiplyScalar(0.3));
                               lDir.normalize();


                               dummy.position.copy(leafBasePos);
                               dummy.lookAt(leafBasePos.clone().add(lDir));
                               dummy.scale.set(leafScale, leafScale, leafScale);
                               dummy.updateMatrix();
                               leafMesh.setMatrixAt(leafIdx++, dummy.matrix);
                           }
                       }
                   }
               }
           }


           plantGroup.add(branchMesh);
           plantGroup.add(leafMesh);
           plantGroup.add(fuzzMesh);


           return plantGroup;
       }


       // Generate and add to scene
       const fern = createProceduralFern();
      
       // Center the fern
       const box = new THREE.Box3().setFromObject(fern);
       const center = box.getCenter(new THREE.Vector3());
       fern.position.sub(new THREE.Vector3(center.x, 0, center.z));
      
       scene.add(fern);


       // Ground plane to ground the shadow
       const groundGeo = new THREE.PlaneGeometry(10, 10);
       const groundMat = new THREE.MeshStandardMaterial({
           color: 0x11140f,
           roughness: 1.0,
           metalness: 0.0
       });
       const ground = new THREE.Mesh(groundGeo, groundMat);
       ground.rotation.x = -Math.PI / 2;
       ground.receiveShadow = true;
       scene.add(ground);


       // --- RENDER LOOP ---
       const clock = new THREE.Clock();


       function animate() {
           requestAnimationFrame(animate);
          
           const delta = clock.getDelta();
          
           // Slowly rotate the entire fern cluster to showcase its 3D volume
           fern.rotation.y += delta * 0.1;


           controls.update();
           renderer.render(scene, camera);
       }


       // Handle Window Resize
       window.addEventListener('resize', () => {
           camera.aspect = window.innerWidth / window.innerHeight;
           camera.updateProjectionMatrix();
           renderer.setSize(window.innerWidth, window.innerHeight);
       });


       animate();
   </script>
