/**
        * Procedurally generates a 3D model of a Hawaiian Kupukupu fern.
        * Returns a Three.js Group containing the entire realistic fern clump.
        */
       function createKupukupuFern() {
           const fernGroup = new THREE.Group();


           // --- MATERIALS ---
           // Stem (Rachis): Slightly brownish-green and somewhat smooth
           const stemMaterial = new THREE.MeshStandardMaterial({
               color: 0x5e482b,
               roughness: 0.8
           });


           // Leaflets (Pinnae): Vibrant, slightly matte fern green
           const leafMaterial = new THREE.MeshStandardMaterial({
               color: 0x4a7c29,
               roughness: 0.6
           });


           // --- REUSABLE GEOMETRY ---
           // Create a single leaflet geometry once and reuse it hundreds of times for performance.
           // A flattened sphere gives a perfect organic capsule/oblong shape.
           const leafGeo = new THREE.SphereGeometry(1, 6, 6);
           leafGeo.scale(0.35, 0.04, 0.09); // Squeeze into a long, thin, flat leaflet
           leafGeo.translate(0.35, 0, 0);   // Shift so the origin (0,0,0) is exactly at the base of the leaf


           /**
            * Generates a single arching frond with pairs of leaflets along its length.
            * @param {number} lengthScale - Multiplier to vary the size of different fronds.
            */
           function buildFrond(lengthScale) {
               const frondGroup = new THREE.Group();
              
               // Kupukupu have many tightly packed leaflets
               const numSegments = 25 + Math.floor(Math.random() * 20);
              
               // Arch dimensions
               const spread = (1.5 + Math.random() * 1.5) * lengthScale; // How far outward it reaches
               const height = (1.2 + Math.random() * 1.5) * lengthScale; // How high the arch peaks
              
               const curvePoints = [];
               const leafPlacements = [];
              
               // 1. Calculate the math for the curving frond
               for (let i = 0; i <= numSegments; i++) {
                   const t = i / numSegments;
                  
                   // The arch moves outward along Z, and curves up and slightly down along Y
                   const z = t * spread;
                   // sin curve creates a natural rising and slightly drooping arc
                   const y = Math.sin(t * Math.PI * 0.85) * height;
                  
                   const pos = new THREE.Vector3(0, y, z);
                   curvePoints.push(pos);
                  
                   // We don't place leaves at the absolute bottom stalk or the extreme tip
                   if (i > 2 && i < numSegments) {
                       // Leaflets are small at the base, largest in the middle, and tiny at the tip
                       const size = Math.sin(t * Math.PI) * (0.6 + Math.random() * 0.4) * lengthScale;
                      
                       // Calculate the slope (pitch) of the stem at this exact point
                       // so the leaflets angle themselves parallel to the stem.
                       const prevY = Math.sin(((i - 1) / numSegments) * Math.PI * 0.85) * height;
                       const prevZ = ((i - 1) / numSegments) * spread;
                       const dy = y - prevY;
                       const dz = z - prevZ;
                       const pitch = Math.atan2(dy, dz);
                      
                       leafPlacements.push({ pos, pitch, size });
                   }
               }
              
               // 2. Build the central stem (Rachis)
               // Using a CatmullRomCurve3 to create a perfectly smooth, curving tube
               const curve = new THREE.CatmullRomCurve3(curvePoints);
               const stemGeo = new THREE.TubeGeometry(curve, numSegments, 0.015 * lengthScale, 5, false);
               const stemMesh = new THREE.Mesh(stemGeo, stemMaterial);
               stemMesh.castShadow = true;
               frondGroup.add(stemMesh);
              
               // 3. Attach the Leaflet pairs along the stem
               leafPlacements.forEach(placement => {
                   const pairGroup = new THREE.Group();
                   pairGroup.position.copy(placement.pos);
                  
                   // Tilt the entire pair to match the arch of the stem
                   pairGroup.rotation.x = -placement.pitch;
                  
                   // --- Right Leaflet ---
                   const rightLeaf = new THREE.Mesh(leafGeo, leafMaterial);
                   rightLeaf.scale.set(placement.size, placement.size, placement.size);
                   rightLeaf.rotation.y = 0.15; // Sweep forward slightly towards the tip
                   rightLeaf.rotation.z = 0.15; // Tilt up into a slight V-shape
                   rightLeaf.castShadow = true;
                   pairGroup.add(rightLeaf);
                  
                   // --- Left Leaflet ---
                   const leftLeaf = new THREE.Mesh(leafGeo, leafMaterial);
                   leftLeaf.scale.set(placement.size, placement.size, placement.size);
                   leftLeaf.rotation.y = Math.PI - 0.15; // Point the opposite way, sweeping forward
                   leftLeaf.rotation.z = -0.15; // Tilt up into a slight V-shape
                   leftLeaf.castShadow = true;
                   pairGroup.add(leftLeaf);
                  
                   frondGroup.add(pairGroup);
               });
              
               return frondGroup;
           }


           // --- BUILD THE CLUMP ---
           // Ferns grow in dense radiating clumps
           const numFronds = 70 + Math.floor(Math.random() * 30); // Increased frond count further to fill the wider base
          
           for (let i = 0; i < numFronds; i++) {
               // Mix of long mature fronds and short new growth
               const lengthScale = 0.5 + Math.random() * 0.8; // Slightly larger minimum size
              
               const frond = buildFrond(lengthScale);
              
               // Radiate the fronds outward in a 360 degree circle
               const baseAngle = (i / numFronds) * Math.PI * 2 + (Math.random() * 0.2);
               frond.rotation.y = baseAngle;
              
               // Widen the base area even more to increase the ground footprint (increased from 1.5 to 3.5)
               frond.position.x = (Math.random() - 0.5) * 3.5;
               frond.position.z = (Math.random() - 0.5) * 3.5;
              
               fernGroup.add(frond);
           }


           // Scale up slightly to make it feel like a substantial planting reward
           fernGroup.scale.set(1.4, 1.4, 1.4);


           return fernGroup;
       }


       // ============================================================================
       // 🌿 END OF KUPUKUPU FUNCTION
       // ============================================================================




       // --- SCENE SETUP FOR PREVIEW ---
       let scene, camera, renderer, controls;


       function init() {
           // 1. Scene
           scene = new THREE.Scene();
           scene.fog = new THREE.FogExp2(0xa3c497, 0.02); // Soft forest fog


           // 2. Camera
           camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
           camera.position.set(0, 4, 8);


           // 3. Renderer
           renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
           renderer.setSize(window.innerWidth, window.innerHeight);
           renderer.shadowMap.enabled = true;
           renderer.shadowMap.type = THREE.PCFSoftShadowMap;
           document.body.appendChild(renderer.domElement);


           // 4. Lights
           const ambientLight = new THREE.AmbientLight(0xffffff, 0.7);
           scene.add(ambientLight);


           const dirLight = new THREE.DirectionalLight(0xfff7e6, 0.9); // Sunlight
           dirLight.position.set(8, 15, 8);
           dirLight.castShadow = true;
           dirLight.shadow.mapSize.width = 2048;
           dirLight.shadow.mapSize.height = 2048;
           scene.add(dirLight);


           // 5. Ground (Stylized dark forest soil)
           const groundGeo = new THREE.CylinderGeometry(15, 15, 2, 32);
           const groundMat = new THREE.MeshStandardMaterial({
               color: 0x3d3024,
               roughness: 1,
               flatShading: true
           });
           const ground = new THREE.Mesh(groundGeo, groundMat);
           ground.position.y = -1;
           ground.receiveShadow = true;
           scene.add(ground);


           // 6. ADD THE KUPUKUPU FERN!
           const kupukupu = createKupukupuFern();
           scene.add(kupukupu);


           // 7. Controls
           controls = new THREE.OrbitControls(camera, renderer.domElement);
           controls.enableDamping = true;
           controls.dampingFactor = 0.05;
           controls.maxPolarAngle = Math.PI / 2 + 0.05; // Lock camera from going under ground
           controls.minDistance = 2;
           controls.maxDistance = 15;
           controls.target.set(0, 1.5, 0); // Focus on the middle of the fern


           // Handle window resize
           window.addEventListener('resize', onWindowResize, false);


           // Start animation loop
           animate();
       }


       function onWindowResize() {
           camera.aspect = window.innerWidth / window.innerHeight;
           camera.updateProjectionMatrix();
           renderer.setSize(window.innerWidth, window.innerHeight);
       }


       function animate() {
           requestAnimationFrame(animate);
           controls.update();
           renderer.render(scene, camera);
       }


       // Initialize when window loads
       window.onload = init;


   </script>
