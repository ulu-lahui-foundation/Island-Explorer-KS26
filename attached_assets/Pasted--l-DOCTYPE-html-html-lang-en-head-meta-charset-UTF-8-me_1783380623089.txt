    <script type="module">
        import * as THREE from 'three';
        import { OrbitControls } from 'three/addons/controls/OrbitControls.js';
        import { ImprovedNoise } from 'three/addons/math/ImprovedNoise.js';

        // --- Core Setup ---
        const container = document.getElementById('webgl-container');
        const scene = new THREE.Scene();
        scene.background = new THREE.Color(0x87CEEB); 
        scene.fog = new THREE.FogExp2(0x87CEEB, 0.002); 

        const camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 1000);
        
        const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2)); 
        renderer.shadowMap.enabled = true;
        renderer.shadowMap.type = THREE.PCFSoftShadowMap;
        container.appendChild(renderer.domElement);

        const controls = new OrbitControls(camera, renderer.domElement);
        controls.enableDamping = true; 
        controls.dampingFactor = 0.05;
        controls.maxPolarAngle = Math.PI / 2 - 0.05; 
        controls.minDistance = 15; 
        controls.maxDistance = 1200; 
        controls.zoomSpeed = 5.0; 
        controls.minAzimuthAngle = -Math.PI / 2.2; 
        controls.maxAzimuthAngle = Math.PI / 2.2;  

        // --- Celestial Bodies & Starfield ---
        const sunGeo = new THREE.SphereGeometry(25, 32, 32);
        const sunMat = new THREE.MeshBasicMaterial({ color: 0xfff5e6 }); 
        const sunMesh = new THREE.Mesh(sunGeo, sunMat);
        scene.add(sunMesh);

        const moonGeo = new THREE.SphereGeometry(18, 32, 32);
        const moonMat = new THREE.MeshBasicMaterial({ color: 0xddeeff });
        const moonMesh = new THREE.Mesh(moonGeo, moonMat);
        scene.add(moonMesh);

        const starsGeo = new THREE.BufferGeometry();
        const starsPositions = [];
        for(let i = 0; i < 800; i++) {
            let x = (Math.random() - 0.5) * 2000;
            let y = (Math.random() - 0.5) * 2000;
            let z = (Math.random() - 0.5) * 2000;
            if(Math.sqrt(x*x + y*y + z*z) > 600) {
                starsPositions.push(x, y, z);
            }
        }
        starsGeo.setAttribute('position', new THREE.Float32BufferAttribute(starsPositions, 3));
        const starsMat = new THREE.PointsMaterial({ color: 0xffffff, size: 2, transparent: true, opacity: 0 });
        const starsGroup = new THREE.Group();
        starsGroup.add(new THREE.Points(starsGeo, starsMat));
        scene.add(starsGroup);

        // --- Lighting ---
        const ambientLight = new THREE.AmbientLight(0xffffff, 0.4);
        scene.add(ambientLight);

        const dirLight = new THREE.DirectionalLight(0xfff5e6, 1.2); 
        dirLight.castShadow = true;
        dirLight.shadow.mapSize.width = 2048;
        dirLight.shadow.mapSize.height = 2048;
        dirLight.shadow.camera.near = 0.1;
        dirLight.shadow.camera.far = 1500; 
        dirLight.shadow.camera.left = -200;
        dirLight.shadow.camera.right = 200;
        dirLight.shadow.camera.top = 200;
        dirLight.shadow.camera.bottom = -200;
        dirLight.shadow.bias = -0.001; 
        scene.add(dirLight);

        const moonLight = new THREE.DirectionalLight(0x88bbff, 0.0); 
        moonLight.castShadow = true;
        moonLight.shadow.mapSize.width = 2048;
        moonLight.shadow.mapSize.height = 2048;
        moonLight.shadow.camera.near = 0.1;
        moonLight.shadow.camera.far = 1500;
        moonLight.shadow.camera.left = -200;
        moonLight.shadow.camera.right = 200;
        moonLight.shadow.camera.top = 200;
        moonLight.shadow.camera.bottom = -200;
        moonLight.shadow.bias = -0.001;
        scene.add(moonLight);

        // --- Procedural Ahupua'a Terrain Generation ---
        const terrainWidth = 120; 
        const terrainLength = 250; 
        const segmentsW = 128;
        const segmentsL = 256;

        const geometry = new THREE.PlaneGeometry(terrainWidth, terrainLength, segmentsW, segmentsL);
        geometry.rotateX(-Math.PI / 2); 

        const positionAttribute = geometry.attributes.position;
        const colors = [];
        const color = new THREE.Color();
        const noise = new ImprovedNoise();

        for (let i = 0; i < positionAttribute.count; i++) {
            let x = positionAttribute.getX(i);
            let z = positionAttribute.getZ(i);
            let y = 0;

            let normalizedZ = (z + (terrainLength / 2)) / terrainLength; 
            let zFactor = 1 - normalizedZ;

            let baseHeight = 72 * Math.pow(zFactor, 2.0);
            let distFromCenter = Math.abs(x) / (terrainWidth / 2); 
            let cornerTaper = 1.0 - Math.pow(zFactor, 1.5) * Math.pow(distFromCenter, 2.0);
            let ridgeHeight = 42 * Math.pow(distFromCenter, 3.2) * Math.pow(zFactor, 1.4) * cornerTaper;

            let noiseVal = noise.noise(x * 0.05, z * 0.05, 0) * 5;
            noiseVal += noise.noise(x * 0.15, z * 0.15, 0) * 2;
            noiseVal *= zFactor; 

            let meander = Math.sin(normalizedZ * Math.PI * 3) * 5; 
            let distToRiver = Math.abs(x - meander);
            let currentRiverWidth = 3.0 + (normalizedZ * 6.0); 

            let riverInfluence = 0;
            if (distToRiver < currentRiverWidth && normalizedZ < 0.95) {
                riverInfluence = 1 - (distToRiver / currentRiverWidth);
                noiseVal *= (1 - riverInfluence); 
            }

            y = baseHeight + ridgeHeight + noiseVal;
            if (riverInfluence > 0) y -= riverInfluence * 3.5; 

            if (normalizedZ > 0.75) {
                let beachFactor = (normalizedZ - 0.75) / 0.25; 
                let targetBeachY = -3.5;
                if (riverInfluence > 0) targetBeachY = -4.5;
                y = THREE.MathUtils.lerp(y, targetBeachY, beachFactor * beachFactor); 
            }

            positionAttribute.setY(i, y);

            if (riverInfluence > 0.001 && normalizedZ < 0.95) {
                color.setHex(0x2774a8).lerp(new THREE.Color(0x1e5a84), riverInfluence);
            } else if (y > 45) {
                color.setHex(0x1f3d14);
                if (noiseVal > 1) color.offsetHSL(0, 0, -0.03);
            } else if (y > 20) {
                color.setHex(0x3a5f22);
                if (noiseVal > 1) color.offsetHSL(0, 0, -0.05); 
            } else if (normalizedZ > 0.75 && y < 3.5) {
                color.setHex(0xe8ddb5);
            } else {
                color.setHex(0x5c8a33);
            }
            colors.push(color.r, color.g, color.b);
        }

        geometry.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
        geometry.computeVertexNormals();

        const material = new THREE.MeshStandardMaterial({
            vertexColors: true, roughness: 0.8, metalness: 0.1, flatShading: true
        });

        const terrain = new THREE.Mesh(geometry, material);
        terrain.receiveShadow = true;
        terrain.castShadow = true;
        scene.add(terrain);

        // --- Surrounding 3D Island Landmass ---
        const islandGeo = new THREE.PlaneGeometry(1200, 1200, 160, 160);
        islandGeo.rotateX(-Math.PI / 2);
        const bdPos = islandGeo.attributes.position;
        const bdColors = [];
        const bdColor = new THREE.Color();

        for (let i = 0; i < bdPos.count; i++) {
            let x = bdPos.getX(i);
            let z = bdPos.getZ(i);
            let newY = 0;

            let originX = 0;
            let originZ = -80; 
            let distToPeak = Math.sqrt((x - originX)**2 + (z - originZ)**2);
            let maxRadius = 500;
            let islandDome = Math.max(0, 1 - (distToPeak / maxRadius));
            
            let sideTaper = Math.max(0.25, 1 - (Math.abs(x) / 250)); 
            newY = islandDome * (160 * sideTaper); 

            let n = noise.noise(x * 0.015, z * 0.015, 0) * 55;
            n += noise.noise(x * 0.05, z * 0.05, 0) * 15;
            newY += (n * islandDome) * sideTaper; 

            if (z > 20) {
                let centerFocus = Math.max(0, 1 - (Math.abs(x) / 450)); 
                let dropFactor = Math.min(1, (z - 20) / 120); 
                let dropAmount = Math.pow(dropFactor * centerFocus, 1.5) * 250; 
                newY -= dropAmount;
            }
            newY = Math.max(-25, newY);

            if (distToPeak > maxRadius - 80) {
                newY = THREE.MathUtils.lerp(newY, -25, (distToPeak - (maxRadius - 80)) / 80);
            }

            let nearestX = Math.max(-58, Math.min(58, x)); 
            let nearestZ = Math.max(-123, Math.min(123, z));
            let isInsideAhupuaa = (x === nearestX && z === nearestZ);

            let normZ = (nearestZ + 125) / 250;
            let zFactor = 1 - normZ;
            let bHeight = 72 * Math.pow(zFactor, 2.0); 
            let dCenter = Math.abs(nearestX) / 60;
            
            let cornerTaper = 1.0 - Math.pow(zFactor, 1.5) * Math.pow(dCenter, 2.0);
            let rHeight = 42 * Math.pow(dCenter, 3.2) * Math.pow(zFactor, 1.4) * cornerTaper; 
            
            let edgeNoise = noise.noise(nearestX * 0.05, nearestZ * 0.05, 0) * 5;
            edgeNoise += noise.noise(nearestX * 0.15, nearestZ * 0.15, 0) * 2;
            edgeNoise *= zFactor;

            let meander = Math.sin(normZ * Math.PI * 3) * 5; 
            let distToRiver = Math.abs(nearestX - meander);
            let currentRiverWidth = 3.0 + (normZ * 6.0); 
            let riverInfluence = 0;
            if (distToRiver < currentRiverWidth && normZ < 0.95) {
                riverInfluence = 1 - (distToRiver / currentRiverWidth);
                edgeNoise *= (1 - riverInfluence); 
            }

            let edgeHeight = bHeight + rHeight + edgeNoise;
            if (riverInfluence > 0) edgeHeight -= riverInfluence * 3.5; 
            
            if (normZ > 0.75) {
                let beachFactor = (normZ - 0.75) / 0.25;
                let targetBeachY = -3;
                if (riverInfluence > 0) targetBeachY = -4.5;
                edgeHeight = THREE.MathUtils.lerp(edgeHeight, targetBeachY, beachFactor * beachFactor); 
            }

            if (isInsideAhupuaa) {
                newY = edgeHeight - 0.5;
            } else {
                let distToEdge = Math.sqrt((x - nearestX)**2 + (z - nearestZ)**2);
                if (distToEdge < 70) {
                    let blend = distToEdge / 70; 
                    blend = blend * blend * (3 - 2 * blend); 
                    newY = THREE.MathUtils.lerp(edgeHeight, newY, blend);
                }
            }

            bdPos.setY(i, newY);

            if (riverInfluence > 0.001 && normZ < 0.95) {
                bdColor.setHex(0x2774a8).lerp(new THREE.Color(0x1e5a84), riverInfluence);
            } else if (newY > 45) {
                bdColor.setHex(0x1f3d14); 
            } else if (newY > 20) {
                bdColor.setHex(0x3a5f22); 
            } else if (normZ > 0.75 && newY < 3.5) {
                bdColor.setHex(0xe8ddb5); 
            } else if (newY > -2) {
                bdColor.setHex(0x5c8a33); 
            } else {
                bdColor.setHex(0xe8ddb5); 
            }
            bdColors.push(bdColor.r, bdColor.g, bdColor.b);
        }

        islandGeo.setAttribute('color', new THREE.Float32BufferAttribute(bdColors, 3));
        islandGeo.computeVertexNormals();

        const backdropMat = new THREE.MeshStandardMaterial({
            vertexColors: true, roughness: 0.9, metalness: 0.05, flatShading: true
        });

        const backdrop = new THREE.Mesh(islandGeo, backdropMat);
        backdrop.receiveShadow = true;
        backdrop.castShadow = true;
        scene.add(backdrop);

        // --- Surrounding Forests (Outside the Ahupua'a) ---
        const treeCount = 1500; 
        const trunkGeo = new THREE.CylinderGeometry(0.3, 0.5, 2.5, 5);
        const leavesGeo = new THREE.DodecahedronGeometry(2.5, 0); 
        leavesGeo.translate(0, 2.2, 0); 

        const trunkMat = new THREE.MeshStandardMaterial({ color: 0x4a3c31, roughness: 0.9, flatShading: true });
        const leavesMat = new THREE.MeshStandardMaterial({ color: 0xffffff, roughness: 0.8, flatShading: true }); 
        
        const trunkInstanced = new THREE.InstancedMesh(trunkGeo, trunkMat, treeCount);
        const leavesInstanced = new THREE.InstancedMesh(leavesGeo, leavesMat, treeCount);
        trunkInstanced.castShadow = true; trunkInstanced.receiveShadow = true;
        leavesInstanced.castShadow = true; leavesInstanced.receiveShadow = true;

        const dummyTree = new THREE.Object3D();
        const treeColor = new THREE.Color();
        let treesPlaced = 0;

        while(treesPlaced < treeCount) {
            let x = (Math.random() - 0.5) * 1000;
            let z = (Math.random() - 0.5) * 1000;

            let originX = 0, originZ = -80, maxRadius = 500;
            let distToPeak = Math.sqrt((x - originX)**2 + (z - originZ)**2);
            if (distToPeak > maxRadius - 20) continue; 

            let islandDome = Math.max(0, 1 - (distToPeak / maxRadius));
            let sideTaper = Math.max(0.25, 1 - (Math.abs(x) / 250)); 
            let treeY = islandDome * (160 * sideTaper); 

            let n = noise.noise(x * 0.015, z * 0.015, 0) * 55;
            n += noise.noise(x * 0.05, z * 0.05, 0) * 15;
            treeY += (n * islandDome) * sideTaper; 

            if (z > 20) {
                let centerFocus = Math.max(0, 1 - (Math.abs(x) / 450)); 
                let dropFactor = Math.min(1, (z - 20) / 120); 
                treeY -= Math.pow(dropFactor * centerFocus, 1.5) * 250; 
            }
            treeY = Math.max(-25, treeY);
            if (distToPeak > maxRadius - 80) treeY = THREE.MathUtils.lerp(treeY, -25, (distToPeak - (maxRadius - 80)) / 80);

            let nearestX = Math.max(-58, Math.min(58, x)); 
            let nearestZ = Math.max(-123, Math.min(123, z));
            if (x === nearestX && z === nearestZ) continue; 

            let normZ = (nearestZ + 125) / 250;
            let zFactor = 1 - normZ;
            let bHeight = 72 * Math.pow(zFactor, 2.0); 
            let dCenter = Math.abs(nearestX) / 60;
            let cornerTaper = 1.0 - Math.pow(zFactor, 1.5) * Math.pow(dCenter, 2.0);
            let rHeight = 42 * Math.pow(dCenter, 3.2) * Math.pow(zFactor, 1.4) * cornerTaper;
            let edgeNoise = noise.noise(nearestX * 0.05, nearestZ * 0.05, 0) * 5 + noise.noise(nearestX * 0.15, nearestZ * 0.15, 0) * 2;
            edgeNoise *= zFactor;

            let meander = Math.sin(normZ * Math.PI * 3) * 5; 
            let distToRiver = Math.abs(nearestX - meander);
            let currentRiverWidth = 3.0 + (normZ * 6.0); 
            let riverInfluence = 0;
            if (distToRiver < currentRiverWidth && normZ < 0.95) {
                riverInfluence = 1 - (distToRiver / currentRiverWidth);
                edgeNoise *= (1 - riverInfluence); 
            }

            let edgeHeight = bHeight + rHeight + edgeNoise;
            if (riverInfluence > 0) edgeHeight -= riverInfluence * 3.5; 
            if (normZ > 0.75) {
                let beachFactor = (normZ - 0.75) / 0.25;
                edgeHeight = THREE.MathUtils.lerp(edgeHeight, riverInfluence > 0 ? -4.5 : -3, beachFactor * beachFactor); 
            }

            let distToEdge = Math.sqrt((x - nearestX)**2 + (z - nearestZ)**2);
            if (distToEdge < 70) {
                let blend = distToEdge / 70; 
                blend = blend * blend * (3 - 2 * blend); 
                treeY = THREE.MathUtils.lerp(edgeHeight, treeY, blend);
            }

            if (treeY > 2 && treeY < 90) {
                let scale = 0.6 + Math.random() * 0.8;
                dummyTree.position.set(x, treeY - 0.2, z); 
                dummyTree.scale.set(scale, scale * (0.8 + Math.random() * 0.5), scale);
                dummyTree.rotation.set((Math.random()-0.5)*0.2, Math.random()*Math.PI*2, (Math.random()-0.5)*0.2); 
                dummyTree.updateMatrix();
                
                trunkInstanced.setMatrixAt(treesPlaced, dummyTree.matrix);
                leavesInstanced.setMatrixAt(treesPlaced, dummyTree.matrix);
                treeColor.setHex(0x4b7022).offsetHSL((Math.random()-0.5)*0.1, (Math.random()-0.5)*0.15, (Math.random()-0.5)*0.1);
                leavesInstanced.setColorAt(treesPlaced, treeColor);
                treesPlaced++;
            }

                    }

                    if (!isValid) {
                        showNotification(`Cannot place ${itemType.charAt(0).toUpperCase() + itemType.slice(1)} here. It belongs in the ${expectedZone}.`);
                        return;
                    }

                    const itemClass = itemCategory === 'plant' ? '.plant-item' : '.build-item';
                    const itemEl = document.querySelector(`${itemClass}[data-type="${itemType}"]`);
                    if (itemEl) {
                        let count = parseInt(itemEl.dataset.count);
                        if (count > 0) {
                            if (itemCategory === 'plant') {
                                spawnPlant(itemType, point);
                            } else {
                                spawnStructure(itemType, point);
                            }
                            count--;
                            itemEl.dataset.count = count;
                            itemEl.querySelector('.plant-count, .item-count').innerText = count;
                            if (count === 0) {
                                itemEl.classList.add('opacity-50', 'grayscale', 'cursor-not-allowed');
                                itemEl.classList.remove('cursor-grab', 'hover:bg-emerald-100', 'hover:bg-amber-100');
                                itemEl.setAttribute('draggable', 'false');
                            }
                        }
                    }
                }
            }
        });

        // Function to create a slightly scaled up inverted hull for the white outline effect
        function addWhiteOutline(meshGroup) {
            const outlineMat = new THREE.MeshBasicMaterial({ color: 0xffffff, side: THREE.BackSide });
            const meshesToOutline = [];
            
            meshGroup.traverse(child => {
                if (child.isMesh && child.material !== outlineMat) { 
                    meshesToOutline.push(child);
                }
            });
            
            meshesToOutline.forEach(mesh => {
                const outline = new THREE.Mesh(mesh.geometry, outlineMat);
                outline.position.copy(mesh.position);
                outline.rotation.copy(mesh.rotation);
                outline.scale.copy(mesh.scale).multiplyScalar(1.08); 
                mesh.parent.add(outline);
            });
        }

        // --- Animation Loop ---
        const clock = new THREE.Clock();

        function animate() {
            requestAnimationFrame(animate);
            const time = clock.getElapsedTime();

            // --- Day/Night Cycle Engine (Real-time Hawaii Time) ---
            const now = new Date();
            const hawaiiTime = new Date(now.toLocaleString('en-US', { timeZone: 'Pacific/Honolulu' }));
            const hours = hawaiiTime.getHours();
            const minutes = hawaiiTime.getMinutes();
            const seconds = hawaiiTime.getSeconds();
            const ms = hawaiiTime.getMilliseconds();
            
            const currentMs = (hours * 3600 + minutes * 60 + seconds) * 1000 + ms;
            const dayProgress = currentMs / 86400000; 
            const dayTime = (dayProgress * 2 * Math.PI) - (Math.PI / 2);
            const orbitRadius = 220; 

            sunMesh.position.x = Math.cos(dayTime) * orbitRadius;
            sunMesh.position.y = Math.sin(dayTime) * orbitRadius;
            sunMesh.position.z = -220; 
            moonMesh.position.x = Math.cos(dayTime + Math.PI) * orbitRadius;
            moonMesh.position.y = Math.sin(dayTime + Math.PI) * orbitRadius;
            moonMesh.position.z = -220;

            dirLight.position.copy(sunMesh.position);
            moonLight.position.copy(moonMesh.position);
            starsGroup.rotation.z = -dayTime;

            let elevation = Math.sin(dayTime); 
            const noonColor = new THREE.Color(0x87CEEB);
            const sunsetColor = new THREE.Color(0xff8c00);
            const nightColor = new THREE.Color(0x0f172a); 
            const skyColor = new THREE.Color();

            if (elevation > 0.2) {
                skyColor.copy(noonColor);
                dirLight.intensity = THREE.MathUtils.lerp(0.8, 1.2, (elevation - 0.2) / 0.8);
                sunMat.color.setHex(0xfff5e6); 
            } else if (elevation > 0.0) {
                let blend = elevation / 0.2;
                skyColor.copy(sunsetColor).lerp(noonColor, blend);
                dirLight.intensity = blend * 0.8;
                sunMat.color.setHex(0xff5500).lerp(new THREE.Color(0xfff5e6), blend); 
            } else if (elevation > -0.2) {
                let blend = (elevation + 0.2) / 0.2;
                skyColor.copy(nightColor).lerp(sunsetColor, blend);
                dirLight.intensity = 0; 
                sunMat.color.setHex(0xff0000).lerp(new THREE.Color(0xff5500), blend); 
            } else {
                skyColor.copy(nightColor);
                dirLight.intensity = 0;
            }

            moonLight.intensity = Math.max(0, -elevation * 0.7); 
            ambientLight.intensity = THREE.MathUtils.lerp(0.3, 0.5, Math.max(0, elevation)); 
            starsMat.opacity = Math.max(0, -elevation * 1.5); 

            scene.background = skyColor;
            scene.fog.color = skyColor;

            // Animate Ocean Waves
            const oceanPositions = ocean.geometry.attributes.position;
            for (let i = 0; i < oceanPositions.count; i++) {
                const x = oceanInitialPositions[i * 3];
                const z = oceanInitialPositions[i * 3 + 2];
                const y = Math.sin(x * 0.02 + time) * 0.8 + Math.cos(z * 0.02 + time) * 0.8;
                oceanPositions.setY(i, y);
            }
            ocean.geometry.attributes.position.needsUpdate = true;

            // Animate Clouds
            clouds.forEach(cloudObj => {
                cloudObj.mesh.position.x += cloudObj.speed;
                if (cloudObj.mesh.position.x > 600) {
                    cloudObj.mesh.position.x = -600;
                    cloudObj.mesh.position.y = 120 + Math.random() * 60;
                    cloudObj.mesh.position.z = (Math.random() - 0.5) * 1000;
                }
            });

            // Animate Birds
            birds.forEach(birdObj => {
                const flapAngle = Math.sin(time * birdObj.flapSpeed) * 0.6;
                birdObj.rightWing.rotation.z = flapAngle;
                birdObj.leftWing.rotation.z = -flapAngle;
                birdObj.mesh.translateZ(birdObj.speed);
                birdObj.mesh.rotation.y += birdObj.turnSpeed;
                birdObj.mesh.position.y = birdObj.baseY + Math.sin(time * 2 + birdObj.offset) * 2;
                if (Math.random() < 0.01) birdObj.turnSpeed = (Math.random() - 0.5) * 0.02;
                const bPos = birdObj.mesh.position;
                if (bPos.x > 120 || bPos.x < -120 || bPos.z > 150 || bPos.z < -150) {
                    birdObj.mesh.lookAt(0, birdObj.mesh.position.y, 0);
                }
            });

            // Animate Sea Animals
            seaAnimals.forEach(animal => {
                animal.mesh.translateZ(animal.speed);
                animal.mesh.rotation.y += animal.turnSpeed;
                if (animal.type === 'honu') {
                    animal.animCycle += 0.05;
                    animal.mesh.position.y = animal.baseY + Math.sin(time * 0.5) * 0.5;
                    animal.flipperL.rotation.z = Math.sin(animal.animCycle) * 0.3;
                    animal.flipperR.rotation.z = -Math.sin(animal.animCycle) * 0.3;
                } else if (animal.type === 'naia') {
                    animal.animCycle += animal.speed * 0.2;
                    let yOffset = Math.sin(animal.animCycle) * 2;
                    animal.mesh.position.y = animal.baseY + yOffset;
                    animal.mesh.rotation.x = -Math.cos(animal.animCycle) * 0.4; 
                }
                const pos = animal.mesh.position;
                if (pos.x > 250 || pos.x < -250 || pos.z < 100 || pos.z > 500) animal.mesh.lookAt(0, animal.mesh.position.y, 300);
                if (Math.random() < 0.01) animal.turnSpeed = (Math.random() - 0.5) * 0.02;
            });

            // Animate Reef Fish Schools
            fishSchools.forEach(school => {
                school.group.translateZ(school.speed);
                school.group.rotation.y += school.turnSpeed;
                school.animCycle += 0.15;
                school.group.position.y = school.baseY + Math.sin(time * 0.8) * 0.3;
                school.fishes.forEach(f => {
                    f.mesh.rotation.y = Math.sin(school.animCycle + f.offset) * 0.3;
                    f.mesh.position.x = f.baseX + Math.sin(school.animCycle * 0.5 + f.offset) * 0.5;
                });
                const pos = school.group.position;
                if (pos.x > 120 || pos.x < -120 || pos.z < 95 || pos.z > 220) school.group.lookAt(0, school.group.position.y, 140);
                if (Math.random() < 0.02) school.turnSpeed = (Math.random() - 0.5) * 0.02;
            });

            // Animate River Particles
            const rPos = riverParticles.geometry.attributes.position.array;
            for (let i = 0; i < riverParticleCount; i++) {
                let z = rPos[i * 3 + 2] + riverParticleData[i].speed;
                if (z > 95) z = -120;
                let normZ = (z + 125) / 250;
                let currentRiverWidth = 3.0 + (normZ * 6.0);
                let meander = Math.sin(normZ * Math.PI * 3) * 5;
                let x = meander + (riverParticleData[i].offset * currentRiverWidth);
                let zFactor = 1 - normZ;
                let y = 72 * Math.pow(zFactor, 2.0);
                if (normZ > 0.75) {
                    let beachFactor = (normZ - 0.75) / 0.25;
                    y = THREE.MathUtils.lerp(y, -4.5, beachFactor * beachFactor);
                } else {
                    y -= 2.8; 
                }
                rPos[i * 3] = x;
                rPos[i * 3 + 1] = y + Math.sin(time * 5 + i) * 0.2; 
                rPos[i * 3 + 2] = z;
            }
            riverParticles.geometry.attributes.position.needsUpdate = true;

            // Animate Ocean Spray Particles
            const oPos = oceanParticles.geometry.attributes.position.array;
            for (let i = 0; i < oceanParticleCount; i++) {
                let origX = oceanParticleData[i].x;
                let origZ = oceanParticleData[i].z;
                origZ -= 0.08;
                if (origZ < 85) origZ = 485; 
                oceanParticleData[i].z = origZ;
                let waveY = Math.sin(origX * 0.02 + time) * 0.8 + Math.cos(origZ * 0.02 + time) * 0.8;
                let foamPop = Math.sin(time * 3 + oceanParticleData[i].phase);
                let y = waveY + (foamPop > 0.8 ? (foamPop - 0.8) * 2 : 0) - 0.2;
                oPos[i * 3] = origX;
                oPos[i * 3 + 1] = y;
                oPos[i * 3 + 2] = origZ;
            }
            oceanParticles.geometry.attributes.position.needsUpdate = true;

            controls.target.x = THREE.MathUtils.clamp(controls.target.x, -60, 60);
            controls.target.y = THREE.MathUtils.clamp(controls.target.y, -10, 100);
            controls.target.z = THREE.MathUtils.clamp(controls.target.z, -80, 150);
            controls.update(); 

            renderer.render(scene, camera);
        }

        window.onload = function() {
            animate();
        }

    </script>
