import * as THREE from 'three';


/**
* Procedurally generates a 3D model of a Hawaiian Taro (Kalo) plant.
* Returns a self-contained THREE.Group containing the entire plant hierarchy.
* Built entirely by clustering simple geometries (Spheres, Cylinders) to form
* complex organic structures like the distinctive sagittate leaves and corm.
* * @returns {THREE.Group} The generated taro plant model.
*/
export function generateProceduralTaro() {
   const plantGroup = new THREE.Group();
   plantGroup.name = "ProceduralTaroPlant";


   // --- SHARED MATERIALS ---
   // Using MeshStandardMaterial for natural, non-flat shaded realism.
   const matCorm = new THREE.MeshStandardMaterial({
       color: 0x4a3225, // Earthy dark brown/purple
       roughness: 0.9,
       metalness: 0.05
   });


   const matRoot = new THREE.MeshStandardMaterial({
       color: 0x7a5c43, // Lighter tan/brown for roots
       roughness: 0.95,
       metalness: 0.0
   });


   // Taro leaves have a matte, velvety surface that beads water (high roughness)
   const matLeafFlesh = new THREE.MeshStandardMaterial({
       color: 0x366e2d, // Deep vibrant natural green
       roughness: 0.85,
       metalness: 0.05,
       side: THREE.DoubleSide
   });


   const matVein = new THREE.MeshStandardMaterial({
       color: 0x5a9e45, // Lighter, slightly yellowish green for veins
       roughness: 0.7,
       metalness: 0.0
   });


   // Shared simple geometries for clustering
   const sphereGeo = new THREE.SphereGeometry(1, 16, 12);
   const cylinderGeo = new THREE.CylinderGeometry(1, 1, 1, 8);
   cylinderGeo.translate(0, 0.5, 0); // Translate so origin is at the base


   // --- 1. GENERATE THE CORM (Underground Tuber Base) ---
   const cormGroup = new THREE.Group();
   cormGroup.name = "Corm";
  
   // Instead of one sphere, we cluster flattened cylinders to create natural ridges
   const numCormSlices = 10;
   const cormBaseHeight = 0.5;
   let previousRadius = 0;


   for (let i = 0; i < numCormSlices; i++) {
       const t = i / (numCormSlices - 1);
       // Bulbous profile: wider in the middle, tapering at top and bottom
       const radius = Math.sin(t * Math.PI) * 2.5 + 0.8;
      
       const sliceGeo = new THREE.CylinderGeometry(radius * 0.9, radius, cormBaseHeight, 16);
       const slice = new THREE.Mesh(sliceGeo, matCorm);
      
       // Add procedural imperfection (jiggle positions slightly)
       slice.position.set(
           (Math.random() - 0.5) * 0.2,
           i * cormBaseHeight * 0.8, // 0.8 overlap creates continuous surface
           (Math.random() - 0.5) * 0.2
       );
      
       slice.castShadow = true;
       slice.receiveShadow = true;
       cormGroup.add(slice);
       previousRadius = radius;
   }
  
   // --- 2. GENERATE ROOTS ---
   // Cluster squiggly cylinders spreading from the lower half of the corm
   const numRoots = 45;
   for (let i = 0; i < numRoots; i++) {
       const rootGroup = new THREE.Group();
      
       // Random spherical coordinate on the bottom hemisphere
       const phi = Math.PI / 2 + Math.random() * (Math.PI / 2);
       const theta = Math.random() * Math.PI * 2;
       const radiusC = 2.0;
      
       let currP = new THREE.Vector3(
           radiusC * Math.sin(phi) * Math.cos(theta),
           (radiusC * Math.cos(phi)) + (cormBaseHeight * 3), // offset to lower part of corm slices
           radiusC * Math.sin(phi) * Math.sin(theta)
       );
      
       // Initial outward direction
       let currDir = currP.clone().normalize();
       currDir.y -= 0.5; // push downwards
       currDir.normalize();


       const numSegments = 4 + Math.floor(Math.random() * 3);
       let currentThickness = 0.08;


       for (let k = 0; k < numSegments; k++) {
           const segLen = 0.4 + Math.random() * 0.5;
           const nextP = currP.clone().add(currDir.clone().multiplyScalar(segLen));
          
           const nextThickness = currentThickness * 0.7;
           const rGeo = new THREE.CylinderGeometry(nextThickness, currentThickness, segLen, 5);
           const rMesh = new THREE.Mesh(rGeo, matRoot);
          
           // Position at midpoint
           rMesh.position.copy(currP.clone().lerp(nextP, 0.5));
           rMesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), currDir);
          
           rootGroup.add(rMesh);
          
           currP = nextP;
           currentThickness = nextThickness;
           // Introduce procedural wiggle
           currDir.add(new THREE.Vector3((Math.random()-0.5)*0.8, (Math.random()-0.5)*0.4, (Math.random()-0.5)*0.8)).normalize();
       }
       cormGroup.add(rootGroup);
   }
   plantGroup.add(cormGroup);


   // --- 3. HELPER: GENERATE PROCEDURAL LEAF ---
   // Builds a complex leaf by clustering flattened spheres for the flesh and thin cylinders for veins
   function buildTaroLeaf(length, width) {
       const leafGroup = new THREE.Group();
      
       const heightLobe = length * 0.35;
       const numBodySteps = 16;
       const numLobeSteps = 8;


       // Function to create a "cupped" or drooping curvature across the leaf surface
       const getZOffset = (x, y) => {
           // Parabolic bowl opening downwards
           return -0.04 * (x * x + Math.pow(Math.max(0, -y), 1.3));
       };


       // A. Build Main Body (Descending from notch/attachment to tip)
       for (let i = 0; i <= numBodySteps; i++) {
           const t = i / numBodySteps; // 0 to 1
           const y = -t * length;
          
           // Procedural width profile: Sagittate (arrow/heart) shape, wide near top, tapering to tip
           const widthF = (1 - Math.pow(t, 1.4)) * width;
          
           // Cluster a flattened sphere for this slice of the leaf flesh
           const flesh = new THREE.Mesh(sphereGeo, matLeafFlesh);
           const zOffset = getZOffset(0, y) + i * 0.002; // Tiny Z increment prevents z-fighting
          
           flesh.position.set(0, y, zOffset);
           // Scale sphere into a flat oval to form part of the leaf blade
           flesh.scale.set(widthF * 0.5, (length / numBodySteps) * 1.8, 0.06);
           flesh.castShadow = true; flesh.receiveShadow = true;
           leafGroup.add(flesh);


           // Center Vein
           if (i < numBodySteps) {
               const nextY = -((i + 1) / numBodySteps) * length;
               const nextZ = getZOffset(0, nextY);
               const veinLen = y - nextY;
              
               const vRadius1 = 0.18 * (1 - t) + 0.03;
               const vRadius2 = 0.18 * (1 - ((i + 1) / numBodySteps)) + 0.03;
              
               const vein = new THREE.Mesh(new THREE.CylinderGeometry(vRadius2, vRadius1, veinLen * 1.1, 6), matVein);
               vein.position.set(0, (y + nextY) / 2, (zOffset + nextZ) / 2 + 0.08); // Float slightly above flesh
               leafGroup.add(vein);
           }


           // Lateral Veins (radiating outwards)
           if (i > 1 && i < numBodySteps - 2 && i % 2 === 0) {
               const veinSpread = widthF * 0.85;
               const latLen = Math.sqrt(Math.pow(veinSpread / 2, 2) + Math.pow(length * 0.15, 2));
               const vThickness = 0.06 * (1 - t) + 0.02;


               // Left lateral
               const leftVein = new THREE.Mesh(new THREE.CylinderGeometry(0.01, vThickness, latLen, 5), matVein);
               leftVein.position.set(-veinSpread / 4, y - length * 0.075, zOffset + 0.06);
               leftVein.rotation.z = Math.atan2(length * 0.15, veinSpread / 2);
               leafGroup.add(leftVein);


               // Right lateral
               const rightVein = new THREE.Mesh(new THREE.CylinderGeometry(0.01, vThickness, latLen, 5), matVein);
               rightVein.position.set(veinSpread / 4, y - length * 0.075, zOffset + 0.06);
               rightVein.rotation.z = -Math.atan2(length * 0.15, veinSpread / 2);
               leafGroup.add(rightVein);
           }
       }


       // B. Build Top Lobes (Growing upwards and outwards from the notch)
       for (let side of [-1, 1]) {
           for (let i = 0; i <= numLobeSteps; i++) {
               const t = i / numLobeSteps;
               const x = side * t * width * 0.42;
               const y = t * heightLobe;
              
               // Lobe tapers elegantly
               const widthF = (1 - Math.pow(t, 1.8)) * (width * 0.45);
              
               const flesh = new THREE.Mesh(sphereGeo, matLeafFlesh);
               const zOffset = getZOffset(x, y) + i * 0.002 - 0.02; // Keep slightly below center node
              
               flesh.position.set(x, y, zOffset);
               flesh.rotation.z = side * -Math.PI / 5 * t; // Angle spheres to follow the organic curve
               flesh.scale.set(widthF, (heightLobe / numLobeSteps) * 2.0, 0.06);
               flesh.castShadow = true; flesh.receiveShadow = true;
               leafGroup.add(flesh);


               // Main lobe vein
               if (i < numLobeSteps) {
                   const nextT = (i + 1) / numLobeSteps;
                   const nextX = side * nextT * width * 0.42;
                   const nextY = nextT * heightLobe;
                  
                   const vLen = Math.sqrt((nextX - x)**2 + (nextY - y)**2);
                   const vRadius1 = 0.12 * (1 - t) + 0.02;
                   const vRadius2 = 0.12 * (1 - nextT) + 0.02;
                  
                   const vein = new THREE.Mesh(new THREE.CylinderGeometry(vRadius2, vRadius1, vLen * 1.1, 6), matVein);
                   vein.position.set((x + nextX) / 2, (y + nextY) / 2, zOffset + 0.08);
                   vein.rotation.z = Math.atan2(y - nextY, x - nextX) + Math.PI / 2;
                   leafGroup.add(vein);
               }
           }
       }
      
       return leafGroup;
   }


   // --- 4. GENERATE STALKS (Hā) AND ATTACH LEAVES ---
   const numStalks = 6 + Math.floor(Math.random() * 3);
   const topOfCorm = numCormSlices * cormBaseHeight * 0.8;


   for (let i = 0; i < numStalks; i++) {
       // Base procedural variables for each stalk
       const isCenterSprout = (i === 0); // Make the first one a central, unrolling shoot
      
       const angle = (i / numStalks) * Math.PI * 2 + (Math.random() * 0.5);
       const stalkHeight = isCenterSprout ? 28 : 20 + Math.random() * 10;
       const stalkLean = isCenterSprout ? 2 : 12 + Math.random() * 8;
      
       // Define stalk curve using Quadratic Bezier points
       const p0 = new THREE.Vector3(
           Math.cos(angle) * 0.5,
           topOfCorm,
           Math.sin(angle) * 0.5
       );
       const p1 = new THREE.Vector3(
           Math.cos(angle) * stalkLean * 0.5,
           topOfCorm + stalkHeight * 0.4,
           Math.sin(angle) * stalkLean * 0.5
       );
       const p2 = new THREE.Vector3(
           Math.cos(angle) * stalkLean,
           topOfCorm + stalkHeight,
           Math.sin(angle) * stalkLean
       );


       const curve = new THREE.QuadraticBezierCurve3(p0, p1, p2);
      
       // Construct Stalk by clustering cylinders
       const stalkSegments = 12;
       const stalkGroup = new THREE.Group();
      
       for (let j = 0; j < stalkSegments; j++) {
           const t1 = j / stalkSegments;
           const t2 = (j + 1) / stalkSegments;
          
           const pt1 = curve.getPoint(t1);
           const pt2 = curve.getPoint(t2);
          
           // Stalk tapers upwards
           const radius1 = THREE.MathUtils.lerp(1.8, 0.4, t1);
           const radius2 = THREE.MathUtils.lerp(1.8, 0.4, t2);
           const dist = pt1.distanceTo(pt2);
          
           // Procedural Color gradient along stalk (purplish base to light green top)
           const stalkMat = new THREE.MeshStandardMaterial({
               color: new THREE.Color().lerpColors(
                   new THREE.Color(0x6b4f62), // Purplish base common in Taro
                   new THREE.Color(0x84b55e), // Fresh green top
                   t1
               ),
               roughness: 0.6,
               metalness: 0.05
           });


           const segGeo = new THREE.CylinderGeometry(radius2, radius1, dist * 1.05, 12);
           const seg = new THREE.Mesh(segGeo, stalkMat);
          
           seg.position.copy(pt1.clone().lerp(pt2, 0.5));
           const dir = pt2.clone().sub(pt1).normalize();
           seg.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), dir);
          
           // Flatten the very base segments slightly to wrap around the center
           if (j < 3) {
               seg.scale.set(1.2, 1, 0.8);
           }
          
           seg.castShadow = true; seg.receiveShadow = true;
           stalkGroup.add(seg);
       }
      
       plantGroup.add(stalkGroup);


       // --- 5. ATTACH LEAF TO STALK ---
       if (isCenterSprout) {
           // Create an unrolled/new leaf by clustering tight, upright cylinders
           const sproutGroup = new THREE.Group();
           sproutGroup.position.copy(p2);
           const sproutDir = curve.getTangent(1).normalize();
           sproutGroup.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), sproutDir);
          
           const sMesh = new THREE.Mesh(
               new THREE.CylinderGeometry(0.05, 0.4, 8, 8),
               new THREE.MeshStandardMaterial({color: 0x98d46a, roughness: 0.7})
           );
           sMesh.position.y = 4; // Shift up from origin
           sproutGroup.add(sMesh);
           plantGroup.add(sproutGroup);
       } else {
           // Create a fully open leaf
           const leafLength = 16 + Math.random() * 6;
           const leafWidth = 10 + Math.random() * 5;
           const leaf = buildTaroLeaf(leafLength, leafWidth);
          
           leaf.position.copy(p2);
          
           // Orient leaf accurately: it hangs off the end of the stalk
           const stalkTangent = curve.getTangent(1).normalize();
          
           // 'lookAt' targets a point forward along tangent, and slightly downward
           const droopVector = new THREE.Vector3(0, -1.2, 0);
           const targetPos = p2.clone().add(stalkTangent).add(droopVector);
           leaf.lookAt(targetPos);
          
           // Because our leaf was built flat on XY plane with tip facing -Y,
           // we rotate X by -90deg so the -Y axis points towards the lookAt target (Z-axis).
           leaf.rotateX(-Math.PI / 2);
          
           // Add a tiny random roll rotation for organic variation
           leaf.rotateY((Math.random() - 0.5) * 0.4);


           plantGroup.add(leaf);
       }
   }


   // Optional: center the entire plant group vertically slightly
   plantGroup.position.y = -topOfCorm * 0.5;


   return plantGroup;
}
