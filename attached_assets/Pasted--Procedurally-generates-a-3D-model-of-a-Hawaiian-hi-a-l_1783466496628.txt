

       /**
        * Procedurally generates a 3D model of a Hawaiian ʻōhiʻa lehua tree.
        * Returns a Three.js Group containing the entire low-poly, stylized tree.
        */
       function createOhiaTree() {
           // Main group that will hold the entire tree
           const treeGroup = new THREE.Group();


           // --- MATERIALS ---
           // Trunk: Smoother, more natural greyish-brown. Flat shading removed.
           const trunkMaterial = new THREE.MeshStandardMaterial({
               color: 0x8a8377,
               roughness: 1.0
           });


           // Foliage: Natural, slightly glossy dark green
           const leafMaterial = new THREE.MeshStandardMaterial({
               color: 0x2b3d26,
               roughness: 0.9
           });


           // Blossoms: Deep, vibrant red for the stamens
           const blossomMaterial = new THREE.MeshStandardMaterial({
               color: 0xe61515,
               roughness: 0.4
           });


           // --- PROCEDURAL GENERATION RULES ---
           const maxDepth = 4; // Reduced from 5 to decrease overall branch density


           /**
            * Recursive function to build the twisted branches of the tree.
            * @param {THREE.Object3D} parent - The group to attach this branch to.
            * @param {number} length - The length of the current branch.
            * @param {number} radius - The thickness of the current branch.
            * @param {number} depth - How many more times the branch will split.
            */
           function buildBranch(parent, length, radius, depth) {
               // Create a container for this branch segment
               const branchGroup = new THREE.Group();
               parent.add(branchGroup);


               // Create the physical branch mesh using a higher-segment cylinder for a smoother, rounder look
               // Geometry: radiusTop, radiusBottom, height, radialSegments
               const branchGeo = new THREE.CylinderGeometry(radius * 0.65, radius, length, 12);
               const branchMesh = new THREE.Mesh(branchGeo, trunkMaterial);
              
               // Cylinders are centered by default, so we move it up by half its length
               // to make its base sit at 0,0,0 relative to its group.
               branchMesh.position.y = length / 2;
              
               // Enable shadows
               branchMesh.castShadow = true;
               branchMesh.receiveShadow = true;
               branchGroup.add(branchMesh);


               // Create an anchor point at the top tip of this branch to spawn children
               const tip = new THREE.Group();
               tip.position.y = length;
               branchGroup.add(tip);


               // Recursion logic: If we haven't reached the end of the tree...
               if (depth > 0) {
                   // Spawn 2 to 4 child branches
                   const numChildren = 2 + Math.floor(Math.random() * 3);
                  
                   for (let i = 0; i < numChildren; i++) {
                       // Calculate dimensions for the child branch
                       const childRadius = radius * 0.7;
                       const childLength = length * (0.6 + Math.random() * 0.3); // 60-90% of parent length
                      
                       // Create gnarled, somewhat erratic angles characteristic of ʻōhiʻa
                       const angleX = (Math.random() - 0.5) * 1.5; // Reduced from 2.2 for less horizontal spread
                       const angleZ = (Math.random() - 0.5) * 1.5; // Reduced from 2.2 for less horizontal spread
                       const angleY = Math.random() * Math.PI * 2; // Random rotation around its own axis


                       // Build the child branch attached to the tip
                       const childBranch = buildBranch(tip, childLength, childRadius, depth - 1);
                       childBranch.rotation.set(angleX, angleY, angleZ);
                   }
               } else {
                   // If we are at the end of the branch (depth == 0), add leaves and flowers
                   addFoliage(tip);
               }


               return branchGroup;
           }


           /**
            * Adds the leaf clusters and realistic stamen blossoms to the tip of a branch.
            * @param {THREE.Object3D} parent - The tip of the branch.
            */
           function addFoliage(parent) {
               // 1. Create multiple smaller, smoother leaf clusters for an organic canopy
               const numLeafClusters = 8 + Math.floor(Math.random() * 6);
               for(let l = 0; l < numLeafClusters; l++) {
                   const leafRadius = 0.4 + Math.random() * 0.6;
                   const leafGeo = new THREE.SphereGeometry(leafRadius, 5, 5);
                   const leafMesh = new THREE.Mesh(leafGeo, leafMaterial);
                  
                   // Offset slightly so it looks like organic clustering
                   leafMesh.position.set(
                       (Math.random() - 0.5) * 2.0, // Tucked in closer to the branch
                       (Math.random() - 0.5) * 1.5,
                       (Math.random() - 0.5) * 2.0  // Tucked in closer to the branch
                   );
                   leafMesh.castShadow = true;
                   parent.add(leafMesh);
               }


               // 2. Add realistic Lehua blossoms (made of dozens of thin radiating stamens)
               const numBlossoms = 4 + Math.floor(Math.random() * 5); // 4 to 8 blossom clusters


               // Create a reusable stamen geometry (very thin cylinder)
               const stamenGeo = new THREE.CylinderGeometry(0.015, 0.015, 0.6, 3);
               // Shift the geometry up so it rotates from the base instead of the center
               stamenGeo.translate(0, 0.3, 0);


               for (let i = 0; i < numBlossoms; i++) {
                   const blossomGroup = new THREE.Group();
                  
                   // Distribute blossoms randomly around the outer edges of the foliage
                   blossomGroup.position.set(
                       (Math.random() - 0.5) * 2.5, // Tucked in closer to match the leaves
                       0.5 + Math.random() * 1.5,
                       (Math.random() - 0.5) * 2.5  // Tucked in closer to match the leaves
                   );
                  
                   // Create the iconic pom-pom effect by radiating 40-60 stamens outward
                   const numStamens = 30 + Math.floor(Math.random() * 20);
                   for (let s = 0; s < numStamens; s++) {
                       const stamen = new THREE.Mesh(stamenGeo, blossomMaterial);
                       // Randomly point the stamen in a spherical direction
                       stamen.rotation.x = Math.random() * Math.PI;
                       stamen.rotation.y = Math.random() * Math.PI * 2;
                       blossomGroup.add(stamen);
                   }


                   parent.add(blossomGroup);
               }
           }


           // Start the recursive tree generation
           // Parent: treeGroup, Initial Length: 3 units, Initial Radius: 0.6 units, Depth: maxDepth
           buildBranch(treeGroup, 4.0, 0.8, maxDepth);


           // Scale the whole tree up slightly to look imposing
           treeGroup.scale.set(1.0, 1.0, 1.0); // Reduced overall scale from 1.2 to 1.0


           return treeGroup;
       }


       // ============================================================================
       // 🌲 END OF TREE FUNCTION
       // ============================================================================




       // --- SCENE SETUP FOR PREVIEW ---
       let scene, camera, renderer, controls;


       function init() {
           // 1. Scene
           scene = new THREE.Scene();
           scene.fog = new THREE.FogExp2(0x7a93a3, 0.015);


           // 2. Camera
           camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
           camera.position.set(0, 10, 25);


           // 3. Renderer
           renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
           renderer.setSize(window.innerWidth, window.innerHeight);
           renderer.shadowMap.enabled = true;
           renderer.shadowMap.type = THREE.PCFSoftShadowMap;
           document.body.appendChild(renderer.domElement);


           // 4. Lights
           const ambientLight = new THREE.AmbientLight(0xffffff, 0.5); // Soft white light
           scene.add(ambientLight);


           const dirLight = new THREE.DirectionalLight(0xfff5e6, 1.0); // Warm sunlight
           dirLight.position.set(15, 25, 10);
           dirLight.castShadow = true;
           dirLight.shadow.mapSize.width = 2048;
           dirLight.shadow.mapSize.height = 2048;
           scene.add(dirLight);


           // 5. Ground (Stylized dark lava rock)
           const groundGeo = new THREE.CylinderGeometry(25, 25, 2, 16);
           const groundMat = new THREE.MeshStandardMaterial({
               color: 0x222222,
               roughness: 1,
               flatShading: true
           });
           const ground = new THREE.Mesh(groundGeo, groundMat);
           ground.position.y = -1;
           ground.receiveShadow = true;
           scene.add(ground);


           // 6. ADD THE ʻŌHIʻA TREE!
           const ohiaTree = createOhiaTree();
           scene.add(ohiaTree);


           // 7. Controls
           controls = new THREE.OrbitControls(camera, renderer.domElement);
           controls.enableDamping = true;
           controls.dampingFactor = 0.05;
           controls.maxPolarAngle = Math.PI / 2 + 0.1; // Don't allow camera to go far below ground
           controls.minDistance = 5;
           controls.maxDistance = 40;
           controls.target.set(0, 5, 0); // Focus slightly up the trunk


           // Handle window resize
           window.addEventListener('resize', onWindowResize, false);


           // Start animation loop
           animate();
       }


       function onWindowResize() {
           camera.aspect = window.innerWidth / window.innerHeight;
           camera.updateProjectionMatrix();
           renderer.setSize(window.innerWidth, window.innerHeight);
       }


       function animate() {
           requestAnimationFrame(animate);
           controls.update(); // required if controls.enableDamping is true
           renderer.render(scene, camera);
       }


       // Initialize when window loads
       window.onload = init;


   </script>
