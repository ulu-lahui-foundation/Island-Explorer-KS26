/**
        * Procedurally generates a 3D model of a Hawaiian Naupaka bush.
        * Returns a Three.js Group containing the entire realistic, sprawling shrub.
        */
       function createNaupakaBush() {
           const bushGroup = new THREE.Group();


           // --- MATERIALS ---
           // Stems: Light grayish-tan, transitioning to a woody texture
           const stemMaterial = new THREE.MeshStandardMaterial({
               color: 0x9c9681,
               roughness: 0.9
           });


           // Leaves: Bright, slightly yellowish green, semi-glossy (succulent-like)
           const leafMaterial = new THREE.MeshStandardMaterial({
               color: 0x6e9438,
               roughness: 0.5
           });


           // Flowers: Pure white, slight subsurface scatter look
           const flowerMaterial = new THREE.MeshStandardMaterial({
               color: 0xffffff,
               roughness: 0.7
           });


           // --- PROCEDURAL GENERATION RULES ---
           const maxDepth = 3; // Keep branching shallow for a dense bush, not a tall tree


           /**
            * Recursive function to build the sprawling branches of the bush.
            */
           function buildStem(parent, length, radius, depth) {
               const stemGroup = new THREE.Group();
               parent.add(stemGroup);


               // Create the physical branch mesh
               const stemGeo = new THREE.CylinderGeometry(radius * 0.7, radius, length, 8);
               const stemMesh = new THREE.Mesh(stemGeo, stemMaterial);
              
               // Shift cylinder up so its base is at the origin
               stemMesh.position.y = length / 2;
               stemMesh.castShadow = true;
               stemMesh.receiveShadow = true;
               stemGroup.add(stemMesh);


               // Anchor point for next branches/leaves
               const tip = new THREE.Group();
               tip.position.y = length;
               stemGroup.add(tip);


               if (depth > 0) {
                   // Sprawling branches: spawn 2 to 4 offshoots
                   const numChildren = 2 + Math.floor(Math.random() * 3);
                  
                   for (let i = 0; i < numChildren; i++) {
                       const childRadius = radius * 0.75;
                       const childLength = length * (0.7 + Math.random() * 0.3);
                      
                       // Naupaka branches spread out wide and grow somewhat erratically
                       const angleSpread = 1.0;
                       const angleX = (Math.random() - 0.5) * angleSpread;
                       const angleZ = (Math.random() - 0.5) * angleSpread;
                       const angleY = Math.random() * Math.PI * 2;


                       const childStem = buildStem(tip, childLength, childRadius, depth - 1);
                       childStem.rotation.set(angleX, angleY, angleZ);
                   }
               } else {
                   // At the ends of the stems, spawn leaves and the signature flowers
                   addFoliageAndFlowers(tip);
               }


               return stemGroup;
           }


           /**
            * Adds succulent leaf rosettes and the signature half-flowers.
            */
           function addFoliageAndFlowers(parent) {
               // 1. Naupaka Leaves (Fleshy, rounded, forming a rosette at the branch tip)
               const numLeaves = 6 + Math.floor(Math.random() * 5);
              
               // Create a base flattened sphere for the fleshy leaf shape
               const baseLeafGeo = new THREE.SphereGeometry(0.5, 8, 8);
               baseLeafGeo.scale(0.8, 1.5, 0.2); // Squeeze into a paddle/spatula shape
               baseLeafGeo.translate(0, 0.75, 0); // Shift so it rotates from the base of the leaf


               for(let l = 0; l < numLeaves; l++) {
                   const leafMesh = new THREE.Mesh(baseLeafGeo, leafMaterial);
                  
                   // Arrange leaves in a circular rosette pattern pointing outward and up
                   const angle = (l / numLeaves) * Math.PI * 2;
                   leafMesh.rotation.y = angle;
                   leafMesh.rotation.x = 0.5 + Math.random() * 0.5; // Lean outwards
                  
                   leafMesh.castShadow = true;
                   parent.add(leafMesh);
               }


               // 2. The Signature Half-Flowers
               // Increased probability to 90% chance and more flowers per cluster
               if (Math.random() > 0.1) {
                   const numFlowers = 2 + Math.floor(Math.random() * 4);
                  
                   // Petal geometry: small, thin, elongated flattened spheres
                   const petalGeo = new THREE.SphereGeometry(0.15, 6, 6);
                   petalGeo.scale(0.5, 2.0, 0.2);
                   petalGeo.translate(0, 0.25, 0); // Shift to base


                   for (let f = 0; f < numFlowers; f++) {
                       const flowerGroup = new THREE.Group();
                      
                       // Nestle the flower in the center of the leaf rosette
                       flowerGroup.position.set(
                           (Math.random() - 0.5) * 0.4,
                           0.2 + Math.random() * 0.4,
                           (Math.random() - 0.5) * 0.4
                       );
                      
                       // --- THE HALF-FLOWER LOGIC ---
                       // A normal flower radiates 360 deg. Naupaka only radiates ~180 deg.
                       const numPetals = 5;
                       for (let p = 0; p < numPetals; p++) {
                           const petal = new THREE.Mesh(petalGeo, flowerMaterial);
                          
                           // Spread 5 petals across exactly half a circle (Math.PI)
                           // We offset it slightly so the "flat" side faces a random direction
                           const petalAngle = (p / (numPetals - 1)) * Math.PI;
                          
                           petal.rotation.z = petalAngle - (Math.PI / 2); // Spread like a fan
                           petal.rotation.x = 0.5; // Lean the whole fan outward slightly
                          
                           flowerGroup.add(petal);
                       }


                       // Rotate the entire half-flower group to face a random direction
                       flowerGroup.rotation.y = Math.random() * Math.PI * 2;
                      
                       parent.add(flowerGroup);
                   }
               }
           }


           // --- BUILD THE BUSH ---
           // Instead of one trunk, spawn multiple main stems leaning outward from the origin
           const numMainStems = 5 + Math.floor(Math.random() * 3);
           for (let i = 0; i < numMainStems; i++) {
               // Initial Length: 1.5 to 2.5 units, Initial Radius: 0.3 units
               const length = 1.5 + Math.random() * 1.0;
               const mainStem = buildStem(bushGroup, length, 0.3, maxDepth);
              
               // Spread the base stems around in a circle
               const baseAngle = (i / numMainStems) * Math.PI * 2;
               mainStem.rotation.y = baseAngle;
              
               // Lean them outwards significantly to create a low, wide shrub
               mainStem.rotation.z = 0.4 + Math.random() * 0.6;
           }


           // Adjust overall size
           bushGroup.scale.set(1.5, 1.5, 1.5);


           return bushGroup;
       }


       // ============================================================================
       // 🌿 END OF NAUPAKA FUNCTION
       // ============================================================================




       // --- SCENE SETUP FOR PREVIEW ---
       let scene, camera, renderer, controls;


       function init() {
           // 1. Scene
           scene = new THREE.Scene();
           scene.fog = new THREE.FogExp2(0x8ab6d6, 0.02); // Coastal fog


           // 2. Camera
           camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
           camera.position.set(0, 6, 12);


           // 3. Renderer
           renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
           renderer.setSize(window.innerWidth, window.innerHeight);
           renderer.shadowMap.enabled = true;
           renderer.shadowMap.type = THREE.PCFSoftShadowMap;
           document.body.appendChild(renderer.domElement);


           // 4. Lights
           const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
           scene.add(ambientLight);


           const dirLight = new THREE.DirectionalLight(0xfff7e6, 1.0); // Bright coastal sun
           dirLight.position.set(10, 20, 10);
           dirLight.castShadow = true;
           dirLight.shadow.mapSize.width = 2048;
           dirLight.shadow.mapSize.height = 2048;
           scene.add(dirLight);


           // 5. Ground (Stylized sandy beach to fit Naupaka Kahakai)
           const groundGeo = new THREE.CylinderGeometry(20, 20, 2, 32);
           const groundMat = new THREE.MeshStandardMaterial({
               color: 0xe3d2b1, // Sand color
               roughness: 1,
               flatShading: true
           });
           const ground = new THREE.Mesh(groundGeo, groundMat);
           ground.position.y = -1;
           ground.receiveShadow = true;
           scene.add(ground);


           // 6. ADD THE NAUPAKA BUSH!
           const naupaka = createNaupakaBush();
           scene.add(naupaka);


           // 7. Controls
           controls = new THREE.OrbitControls(camera, renderer.domElement);
           controls.enableDamping = true;
           controls.dampingFactor = 0.05;
           controls.maxPolarAngle = Math.PI / 2 + 0.05; // Lock camera from going under ground
           controls.minDistance = 3;
           controls.maxDistance = 25;
           controls.target.set(0, 2, 0); // Focus on the center of the bush


           // Handle window resize
           window.addEventListener('resize', onWindowResize, false);


           // Start animation loop
           animate();
       }


       function onWindowResize() {
           camera.aspect = window.innerWidth / window.innerHeight;
           camera.updateProjectionMatrix();
           renderer.setSize(window.innerWidth, window.innerHeight);
       }


       function animate() {
           requestAnimationFrame(animate);
           controls.update();
           renderer.render(scene, camera);
       }


       // Initialize when window loads
       window.onload = init;


   </script>
